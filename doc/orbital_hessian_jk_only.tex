\section*{Setup}

Fixed natural occupations $n_p$, $D_{pq}=n_p\delta_{pq}$. A JK-only two-electron
energy functional with a general symmetric occupation-coupling function
$f(n_a,n_b)=f(n_b,n_a)$, $f(n,n)=n$, replacing the two-particle density matrix
elements as
\begin{equation}
\Gamma_{pqpq} = \tfrac12\,n_p n_q \quad (\text{Hartree/direct}), \qquad
\Gamma_{pqqp} = -\tfrac12\,f(n_p,n_q) \quad (\text{exchange}),
\end{equation}
so that
\begin{equation}
E_2 = \tfrac12\sum_{pq} n_p n_q\,\langle pq|pq\rangle
\;-\; \tfrac12\sum_{pq} f(n_p,n_q)\,\langle pq|qp\rangle .
\end{equation}

The M\"uller functional is the special case $f(n_a,n_b)=\sqrt{n_a n_b}$.

\bigskip
\noindent\textbf{Important:} this $\Gamma$ does \emph{not} satisfy the
fermionic antisymmetry $\Gamma_{pqrs}=-\Gamma_{qprs}$ of a genuine two-particle
density matrix (it is an $N$-representability-violating reconstruction, valid
only in the idempotent/integer-occupation limit). The gradient and Hessian
below are therefore derived directly from the explicit energy expression
above (differentiating each integral leg under orbital rotation), \emph{not}
from the operator-commutator machinery used for the general $\Gamma$ case,
since that machinery relied on antisymmetry at two separate steps.

\section*{Gradient}

Define
\begin{equation}
V_{qp} \;\equiv\; h_{qp} + \sum_t n_t\,\langle tq|tp\rangle .
\end{equation}

\begin{equation}
\boxed{\,
g_{pq} \;=\; (n_q-n_p)\,V_{qp}
\;+\; \sum_t \big[f(n_p,n_t)-f(n_q,n_t)\big]\,\langle qt|tp\rangle
\,}
\end{equation}

\section*{Hessian}

\begin{align}
G_{pq,rs} \;=\;\;
& (n_q-n_p)\Big[-\delta_{qr}\,V_{sp} + \delta_{ps}\,V_{qr}
+ (n_s-n_r)\,\langle sq|rp\rangle\Big] \notag\\[4pt]
& -\,\delta_{qr}\sum_t \big[f(n_p,n_t)-f(n_r,n_t)\big]\,\langle st|tp\rangle
\;+\; \delta_{ps}\sum_t \big[f(n_s,n_t)-f(n_q,n_t)\big]\,\langle qt|tr\rangle \notag\\[4pt]
& +\,\big[f(n_p,n_s)-f(n_p,n_r)-f(n_q,n_s)+f(n_q,n_r)\big]\,\langle qs|rp\rangle
\end{align}

\bigskip
\noindent\textbf{Special case (M\"uller):} set $f(n_a,n_b)=\sqrt{n_an_b}$
everywhere above.

\section*{Generalization: independent Hartree coupling}

The derivation only uses that the couplings are fixed, symmetric matrices
attached to orbital positions, so the Hartree factor $n_pn_q$ may be replaced
by an arbitrary symmetric $H_{pq}$ (needed for the antisymmetric M\"uller
variant, whose Hartree coupling is $g(n_p,n_q)=\tfrac12[n_pn_q-\sqrt{n_pn_q}]$,
equal to its exchange coupling), and $f(n_p,n_q)$ by a symmetric $X_{pq}$
(index-dependent couplings such as BBC2 or the diagonal branch of MLSIC/GU need
nothing else). With $E=\sum_a n_ah_{aa}+\tfrac12\sum_{ab}H_{ab}\langle ab|ab\rangle
-\tfrac12\sum_{ab}X_{ab}\langle ab|ba\rangle$, the same leg-by-leg
differentiation gives
\begin{equation}
g_{pq} = (n_q-n_p)\,h_{qp}
+ \sum_t\Big\{\big[H_{qt}-H_{pt}\big]\langle tq|tp\rangle
+ \big[X_{pt}-X_{qt}\big]\langle qt|tp\rangle\Big\},
\end{equation}
\begin{align}
G_{pq,rs} =\;& (n_q-n_p)\big[-\delta_{qr}h_{sp}+\delta_{ps}h_{qr}\big]
- \delta_{qr}\sum_t \big[H_{qt}-H_{pt}\big]\langle ts|tp\rangle
+ \delta_{ps}\sum_t\big[H_{qt}-H_{pt}\big]\langle tq|tr\rangle \notag\\
& -\delta_{qr}\sum_t\big[X_{pt}-X_{qt}\big]\langle st|tp\rangle
+ \delta_{ps}\sum_t\big[X_{pt}-X_{qt}\big]\langle qt|tr\rangle \notag\\
& +\big[H_{qs}-H_{qr}-H_{ps}+H_{pr}\big]\langle sq|rp\rangle
+\big[X_{ps}-X_{pr}-X_{qs}+X_{qr}\big]\langle qs|rp\rangle .
\end{align}
For $H_{ab}=n_an_b$ these are exactly the boxed gradient and the Hessian above
(with $V_{qp}$ expanded and $f\to X$), since $(n_q-n_p)(n_s-n_r)=H_{qs}-H_{qr}-H_{ps}+H_{pr}$.

\section*{Real-step Hessian and code conventions}

$G_{pq,rs}$ is the ``bare'' coefficient, the derivative of $g_{pq}$ along a
rotation generated by $\kappa_{rs}$ (a sequential rotation, which is what a
finite difference of the gradient at a rotated point measures). The Hessian for
the real orbital-rotation step $\kappa_{pq}=+t,\ \kappa_{qp}=-t$ is
\begin{equation}
\mathrm{Hess}_{pq,rs}=\frac{\partial^2E}{\partial\kappa_{pq}\,\partial\kappa_{rs}}
= G_{pq,rs}-G_{pq,sr}-G_{qp,rs}+G_{qp,sr},
\end{equation}
as in \texttt{orbital\_hessian.tex}. In the code
(\texttt{Hessian\_opt/JkOnlyFock.h}, \texttt{JkOnlyHessian.h}) the gradient is
stored with the project convention $g^{\rm code}_{pq}=-2\,g^{*}_{pq}$ (only
$p\ge q$; $\mathrm{Re}\,g^{\rm code}_{pq}=\partial E/\partial\kappa_{pq}$), and
\texttt{jkOnlyHessianElement} returns $\mathrm{Hess}_{pq,rs}$ (its imaginary part
is round-off for physical input).

\section*{Numerical validation}

\noindent\textbf{Status.} The gradient and Hessian above have been checked
numerically against finite differences, for every JK-only functional
implemented in the code (SD, M\"uller, BBC2, CA, CGA, ML, MLSIC, GU, POWER, and the
antisymmetric M\"uller variant), for real orbitals (nonrelativistic) and complex
spinors (X2C and four-component Dirac--Hartree--Fock). They were also re-derived
independently by differentiating the rotated integral legs (as in the
generalization above); this reproduces the boxed expressions term by term.

\medskip
\noindent\textbf{Test protocol.} Orbitals are rotated by $e^{-\kappa}$ with
$\kappa_{rs}=+t,\ \kappa_{sr}=-t$ (one-electron and two-electron integrals
transformed accordingly), at \emph{fixed} occupations, with central steps
$t=\pm10^{-4}$.
\begin{itemize}
\item \emph{Gradient:} $\mathrm{Re}\,g^{\rm code}_{pq}$ against
$[E(+t)-E(-t)]/2t$, with $E$ the energy above at the rotated integrals; and
elementwise against the gradient obtained from the generalized Fock matrix
(an independent route).
\item \emph{Hessian:} $\mathrm{Hess}_{pq,rs}$ against
$[g_{pq}(+t)-g_{pq}(-t)]/2t$, where $g_{pq}(\pm t)$ is the (independently
validated, Fock-matrix based) gradient evaluated at the rotated integrals.
\end{itemize}

\medskip
\noindent\textbf{Example 1: all functionals at generic fractional occupations.}
Water, geometry (\AA) O $(0,0,0)$, H $(0,\pm0.757,0.586)$, STO-3G basis, 10
electrons. Occupations of the active orbitals (all spin-orbitals/spinors of the
positive-energy space) were fixed to the generic values
$n_i=0.15+0.7\,\{0.6180339887\,(i+1)\}$ ($\{\cdot\}$ the fractional part),
i.e.\ strictly inside $(0,1)$ and \emph{not} the result of any optimization, so
that no functional is tested in a trivial near-idempotent limit. The three
orbital pairs with the largest gradient were used automatically; with the two
orientations of $(r,s)$ this gives 18 index quadruples $(p,q,r,s)$ per
functional, including ones sharing an index. Table~\ref{tab:val} lists the
largest deviations for the X2C spinors (for the other two cases see below).
Here $\max|\Delta g|$ is the deviation of the gradient from the energy
finite difference, $\max|\Delta\mathrm{Hess}|$ that of the Hessian from the
finite difference of the gradient (the Hessian elements have magnitudes up to
$\sim 1$).

Hessian is $\lesssim10^{-15}$.

\noindent\textbf{Remark.} Occupations must be genuinely fractional on both
